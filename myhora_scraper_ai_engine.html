ทฤษฎีมาร์คอฟ (Markov Property)

แม้การออกสลากทางทฤษฎีจะเป็นอิสระต่อกัน (Independent Trials) แต่ในทางสถิติประยุกต์ การเก็บสถิติลำดับการเปลี่ยนสถานะ (State Transition) จะช่วยสะท้อนพฤติกรรมการกระจายตัวของอุปกรณ์ออกรางวัลและวงรอบตัวเลขจริง

การเกาะกลุ่มของหลัก (Digit Clustering)

จากการวิเคราะห์ 840+ งวด พบว่า 39.4% ของงวดถัดไปจะมีตัวเลขจากหลักใดหลักหนึ่งของงวดก่อนติดมาด้วย การรู้เลขงวดก่อนหน้าจึงลดความไม่แน่นอน (Entropy) ในการคัดเลขวิ่งเดี่ยวได้อย่างมีนัยสำคัญ

การดึงกลับสู่ค่าเฉลี่ย (Mean Reversion)

หากงวดก่อนหน้าออกเลขที่มีผลรวมแต้มสูงมาก (เช่น 99 = 18 แต้ม) ความน่าจะเป็นของงวดถัดไปจะมีความเอนเอียงตามทฤษฎีที่จะคืนกลับสู่ช่วงแต้มเฉลี่ย (8 - 12 แต้ม) สูงกว่าปกติ

ทำไมดึงข้อมูลจาก MyHora

โครงสร้าง URL ของ MyHora สม่ำเสมอและคาดเดาได้ ย้อนหลังถึง พ.ศ. 2533 (กว่า 34 ปี, 840+ งวด) ดึงได้ 2 แบบ:

รายปี แนะนำ
https://myhora.com/lottery/result-{YYYY}.aspx

เช่น result-2535.aspx จะรวมทุกงวดของปี 2535 (~24 งวด) ไว้ในหน้าเดียว ทำให้ยิง Request เพียง 35 ครั้ง ก็ได้ข้อมูลครบทั้ง 34 ปี (ประหยัดเวลาและ Bandwidth สูงสุด ไม่โดน Rate Limit)

รายงวด
https://myhora.com/lottery/result-{DD}-{MM}-{YYYY}.aspx

เช่น result-16-11-2569.aspx สำหรับเก็บรายละเอียดรางวัลย่อย (รางวัลที่ 2-5, รางวัลข้างเคียง, เลขหน้า 3 ตัว) แบบสมบูรณ์รายงวด

สคริปต์ Python สำหรับดึงข้อมูล

ดึงข้อมูลรายปีอัตโนมัติตั้งแต่ 2533 ถึงปัจจุบัน แล้วบันทึกเป็น JSON

import requests
from bs4 import BeautifulSoup
import json
import time

def scrape_myhora_lottery(start_year_be=2533, end_year_be=2569):
    """
    Scraper ดึงข้อมูลผลสลากกินแบ่งรัฐบาลจาก MyHora แบบรายปี
    ประหยัด Request: วนลูปปี พ.ศ. ดึงครั้งละ 1 ปี (ได้ครบทุกงวดในหน้านั้น)
    """
    headers = {
        'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36'
    }
    all_draws = []

    print(f"[*] เริ่มดึงข้อมูลหวย MyHora ตั้งแต่ปี พ.ศ. {start_year_be} ถึง {end_year_be}...")

    for year in range(start_year_be, end_year_be + 1):
        url = f"https://myhora.com/lottery/result-{year}.aspx"
        try:
            res = requests.get(url, headers=headers, timeout=15)
            if res.status_code != 200:
                print(f"[-] ไม่พบข้อมูลปี {year} (HTTP {res.status_code})")
                continue
            
            soup = BeautifulSoup(res.content, 'html.parser')
            # ตารางผลสลากแต่ละงวดในหน้าสรุปรายปี
            draw_blocks = soup.find_all('div', class_='content-box')
            
            found_count = 0
            for box in draw_blocks:
                header = box.find(['h2', 'h3', 'div'], class_=['title', 'header'])
                if not header:
                    continue
                date_text = header.get_text(strip=True)
                
                # ดึงรางวัลที่ 1 และเลขท้าย 2 ตัว
                prize1 = ""
                two_digit = ""
                
                # ค้นหาค่าจาก class หรือ text ภายใน box
                p1_el = box.find(lambda tag: 'รางวัลที่ 1' in tag.text or 'รางวัลที่1' in tag.text)
                if p1_el:
                    prize1 = p1_el.find_next(['b', 'span', 'div']).get_text(strip=True)
                
                two_el = box.find(lambda tag: 'เลขท้าย 2 ตัว' in tag.text or '2 ตัวท้าย' in tag.text)
                if two_el:
                    two_digit = two_el.find_next(['b', 'span', 'div']).get_text(strip=True)

                if two_digit:
                    all_draws.append({
                        "year_be": year,
                        "date_str": date_text,
                        "first_prize": prize1,
                        "two_digit": two_digit[-2:]
                    })
                    found_count += 1

            print(f"[+] ปี {year}: เก็บสำเร็จ {found_count} งวด")
            time.sleep(0.5) # ป้องกันการยิงถี่เกินไป

        except Exception as e:
            print(f"[!] เกิดข้อผิดพลาดในปี {year}: {e}")

    # บันทึกเป็น JSON
    with open("myhora_lottery_data.json", "w", encoding="utf-8") as f:
        json.dump(all_draws, f, ensure_ascii=False, indent=2)

    print(f"[v] เสร็จสมบูรณ์! รวบรวมข้อมูลได้ทั้งหมด {len(all_draws)} งวด บันทึกลง myhora_lottery_data.json")

if __name__ == "__main__":
    scrape_myhora_lottery()
